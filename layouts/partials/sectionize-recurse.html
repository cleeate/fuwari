{{/* Recursive worker for sectionize.html (remark-sectionize). Context: dict "rest" "out" "stack" (slice of dicts {"d": depth, "bq": bool}). A section opened inside a
(e.g. a heading inside a quoted block) is a child of the blockquote in the original's mdast, so it must close BEFORE the blockquote's
— the stack tracks which sections are inside one. The `bq` entries always form a suffix of the stack. */}} {{- $rest := .rest -}} {{- $out := .out -}} {{- $stack := .stack -}} {{- $heads := findRE `]*>.*?` $rest 1 -}} {{- if not $heads -}} {{- /* Trailing content stays INSIDE the still-open sections, except that sections opened inside a blockquote close before its . */ -}} {{- $anyBQ := false -}} {{- range $stack }}{{ if .bq }}{{ $anyBQ = true }}{{ end }}{{ end -}} {{- $parts := split $rest "" -}} {{- if and $anyBQ (gt (len $parts) 1) -}} {{- $out = printf "%s%s" $out (index $parts 0) -}} {{- range $stack }}{{ if .bq }}{{ $out = printf "%s" $out }}{{ end }}{{ end -}} {{- $out = printf "%s%s" $out (delimit (after 1 $parts) "") -}} {{- range $stack }}{{ if not .bq }}{{ $out = printf "%s" $out }}{{ end }}{{ end -}} {{- return $out -}} {{- end -}} {{- $out = printf "%s%s" $out $rest -}} {{- range $stack }}{{ $out = printf "%s" $out }}{{ end -}} {{- return $out -}} {{- end -}} {{- $h := index $heads 0 -}} {{- $parts := split $rest $h -}} {{- $before := index $parts 0 -}} {{- /* split/join are inverse; substr+len would mix byte and rune offsets */ -}} {{- $after := delimit (after 1 $parts) $h -}} {{- $levelStr := index (findRE `[1-6]` $h 1) 0 -}} {{- $d := int $levelStr -}} {{- /* Is this heading inside a blockquote? Count unclosed ones before it. */ -}} {{- $prefix := printf "%s%s" $out $before -}} {{- $inBQ := gt (sub (len (findRE "
" $prefix)) (len (findRE "
" $prefix))) 0 -}} {{- /* Sections to close: depth >= the new heading's, plus every section inside a blockquote when the new heading is outside it. */ -}} {{- $newStack := slice -}} {{- $closeBQ := 0 -}} {{- $closeTop := 0 -}} {{- range $stack -}} {{- $close := false -}} {{- if ge .d $d }}{{ $close = true }}{{ end -}} {{- if and .bq (not $inBQ) }}{{ $close = true }}{{ end -}} {{- if $close -}} {{- if .bq }}{{ $closeBQ = add $closeBQ 1 }}{{ else }}{{ $closeTop = add $closeTop 1 }}{{ end -}} {{- else -}} {{- $newStack = $newStack | append . -}} {{- end -}} {{- end -}} {{- /* Content before the heading belongs to the open section(s); sections inside a blockquote close before its when one appears here. */ -}} {{- $bparts := split $before "" -}} {{- if and (gt $closeBQ 0) (gt (len $bparts) 1) -}} {{- $out = printf "%s%s" $out (index $bparts 0) -}} {{- range seq $closeBQ }}{{ $out = printf "%s" $out }}{{ end -}} {{- $out = printf "%s%s" $out (delimit (after 1 $bparts) "") -}} {{- range seq $closeTop }}{{ $out = printf "%s" $out }}{{ end -}} {{- else -}} {{- $out = printf "%s%s" $out $before -}} {{- range seq (add $closeBQ $closeTop) }}{{ $out = printf "%s" $out }}{{ end -}} {{- end -}} {{- $out = printf "%s
%s" $out $h -}} {{- $newStack = $newStack | append (dict "d" $d "bq" $inBQ) -}} {{- return partial "sectionize-recurse.html" (dict "rest" $after "out" $out "stack" $newStack) -}}
