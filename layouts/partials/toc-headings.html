{{/* Extract headings from rendered HTML for the TOC (the original's rehype-slug ids + autolink anchors are already in the HTML). Returns a slice of dicts "level" "id" "title"; the title keeps the trailing "#" from the anchor icon, like .Fragments.Headings does — toc.html strips it. */}} {{- $list := slice -}} {{- range findRE `]*id="[^"]+">.*?` . -}} {{- $level := int (index (findRE `[1-6]` . 1) 0) -}} {{- $id := replaceRE `^]*id="([^"]+)"[^>]*>.*$` "$1" . -}} {{- $inner := replaceRE `^]*>(.*)$` "$1" . -}} {{- /* The heading HTML carries Goldmark's entity escaping (& for &); decode it so toc.html's {{ }} escaping produces the same text once, not a double-escaped &amp;. & is decoded last so &lt; survives as <. */ -}} {{- $title := replaceRE `<[^>]+>` "" $inner -}} {{- $title = replaceRE `<` "<" $title -}} {{- $title = replaceRE `>` ">" $title -}} {{- $title = replaceRE `"` `"` $title -}} {{- $title = replaceRE `'` "'" $title -}} {{- $title = replaceRE `&` "&" $title -}} {{- $title = trim $title " " -}} {{- $list = $list | append (dict "level" $level "id" $id "title" $title) -}} {{- end -}} {{- return $list -}}
