{{/* Replaces ::github{repo="owner/repo"} directives with repo cards, one at a time. Context: dict "html" "n" (card counter). Goldmark escapes the quotes in the paragraph text as ", so the pattern matches that form. */}} {{- $html := .html -}} {{- $n := .n -}} {{- $m := findRE `

::github\{repo="([^&]+)"\}

` $html 1 -}} {{- $result := $html -}} {{- if $m -}} {{- $tag := index $m 0 -}} {{- $repo := index (findRE `repo="([^&]+)"` $tag 1) 0 | replaceRE `repo="([^&]+)"` "$1" -}} {{- $id := printf "GC%06d" $n -}} {{- $card := partial "github-card.html" (dict "repo" $repo "id" $id) -}} {{- $parts := split $html $tag -}} {{- $before := index $parts 0 -}} {{- $after := delimit (after 1 $parts) $tag -}} {{- $result = partial "github-cards.html" (dict "html" (printf "%s%s%s" $before $card $after) "n" (add $n 1)) -}} {{- end -}} {{- return $result -}}
